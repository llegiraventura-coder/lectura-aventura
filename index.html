// PARSER CORREGIT (SENSE ICONA I AMB SCROLL GARANTIT) function parseContentWithGlossary(rawText, glossaryList = []) { if (!rawText) return ''; let processed = rawText; // 1. Reemplaça marques explicites [paraula](#glossari-section) processed = processed.replace(/\[([^\]]+)\]\(#glossari-section\)/g, function(match, word) { return `${word}`; }); // 2. Detecció automàtica de paraules del glossari (sense icona de llibret) if (Array.isArray(glossaryList) && glossaryList.length > 0) { glossaryList.forEach(g => { if (g.word && g.word.trim()) { const wordToFind = g.word.trim(); const regex = new RegExp(`\\b(${wordToFind})\\b(?![^<]*>|[^[]*])`, 'gi'); processed = processed.replace(regex, `$1`); } }); } return processed; } // FUNCIÓ AUXILIAR PER ASSEGURAR EL DESPLAÇAMENT SUAU function scrollToGlossary(e) { if (e) e.preventDefault(); const target = document.getElementById('glossary-section'); if (target) { target.scrollIntoView({ behavior: 'smooth', block: 'start' }); } }
